function openURLAtSpecificTime(url, hour, minute, second) {
    const now = new Date();
    const targetTime = new Date(
        now.getFullYear(),
        now.getMonth(),
        now.getDate()+1,
        hour,
        minute,
        second
    );

    let delay = targetTime - now;

    // 20成功,380偷跑太多反而失敗
    // 9點雙tab 20, 280
    delay = delay - 20

    if (delay > 0) {
        setTimeout(function () {
            window.location.href = url;
        }, delay);
    }
}
