//[2,5,8,11]
//[3,6,9,12]

function openURLAtSpecificTime(url, hour, minute, second) {
    const now = new Date();
    const targetTime = new Date(
        now.getFullYear(),
        now.getMonth(),
        now.getDate()+1,
        hour,
        minute,
        second
    );

    let delay = targetTime - now;

    console.log(delay)
    delay = delay - 50

    if (delay > 0) {
        setTimeout(function () {
            window.location.href = url;
        }, delay);
    }
}

// Define the URL and the target time (00:00:00)

const targetHour = 00;
const targetMinute = 00;
const targetSecond = 00;

// Call the function to open the URL at the specified time
